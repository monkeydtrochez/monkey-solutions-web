```typescript
"use client";
import { useState, useEffect, useContext, useMemo } from "react";
import GlobalContext from "@/app/context/GlobalContext";
import { Badge } from "@/components/ui/badge";
import type { Project } from "@/app/models/sanityTypes";

type Filter = "all" | "web" | "ios" | "saas";

const matchesFilter = (kind: string | undefined | null, filter: Filter): boolean => {
  const k = kind ?? "";
  if (filter === "all") return true;
  if (filter === "web") return /commerce|web|booking/i.test(k);
  if (filter === "ios") return /iOS/.test(k);              // case-sensitive — do NOT add "i" flag
  if (filter === "saas") return /SaaS/i.test(k);
  return true;
};

export default function WorkSection() {
  const ctx = useContext(GlobalContext);
  const projects = ctx?.projects ?? [];

  const [filter, setFilter] = useState<Filter>("all");
  const [openId, setOpenId] = useState<string | null>(null);

  // Default-open: first project in sorted list — set after hydration to avoid race condition
  // (per RESEARCH.md Pitfall 6).
  useEffect(() => {
    if (projects.length > 0 && openId === null) {
      setOpenId(projects[0]._id);
    }
    // eslint-disable-next-line react-hooks/exhaustive-deps
  }, [projects.length]);

  const shown = useMemo(
    () => projects.filter((p) => matchesFilter(p.kind, filter)),
    [projects, filter]
  );

  // If the currently-open row is not in the filtered set, collapse it.
  useEffect(() => {
    if (openId && !shown.some((p) => p._id === openId)) {
      setOpenId(null);
    }
  }, [shown, openId]);

  const handleToggle = (id: string) => {
    setOpenId((prev) => (prev === id ? null : id));
  };

  return (
    <section
      id="work"
      style={{
        padding: "var(--section-py) var(--page-px)",
        background: "var(--ms-bg)",
      }}
    >
      <div style={{ maxWidth: "var(--content-max)", margin: "0 auto" }}>
        {/* regions 2..5 */}
      </div>
    </section>
  );
}
```
